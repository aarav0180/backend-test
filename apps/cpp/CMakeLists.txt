cmake_minimum_required(VERSION 3.20)
project(backend_bench_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(Crow GIT_REPOSITORY https://github.com/CrowCpp/Crow.git GIT_TAG v1.2.1)
FetchContent_MakeAvailable(Crow)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(backend-bench-cpp app.cpp)
target_link_libraries(backend-bench-cpp PRIVATE Crow::Crow nlohmann_json::nlohmann_json)
